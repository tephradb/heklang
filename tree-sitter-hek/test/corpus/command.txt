command PlaceOrder(order_id: Uuid, customer_id: Int, notes: String?) {
  guard @order.placed(order_id), @order.cancelled(order_id)

  fold email_taken: Bool = false
    on @order.placed(email) => true

  fold lifetime_spend: Money(2) = 0
    on @order.placed(customer_id) { total } => lifetime_spend + total
    on @order.cancelled(customer_id) { refund } => lifetime_spend - refund

  let customer = customer_id
  fold blocked: Bool = false
    on @customer.blocked(customer_id: customer) => true

  if blocked {
    return reject CustomerBlocked
  } else if email_taken {
    return invalid("that email has already placed an order")
  } else {
    return
  }
}

command ListItem(item_id: Uuid, tier: Tier) {
  for other_id, other in items {
    if other_id != item_id && other.sku == wanted {
      return reject SkuTaken { sku }
    }
  }

  emit @item.listed {
    item_id,
    item: Item { sku: wanted, price, tags },
    tier,
  }
}

================================================================================
declarations interleave with statements, and each run is one read
================================================================================

command ListItem(item_id: Uuid, sku: String) {
  let trimmed = sku.trim()
  if trimmed.is_empty() {
    return invalid("a sku is required")
  }

  guard SkuIsFree { sku: trimmed }

  fold listed: Bool = false
    on @item.listed(item_id) => true

  emit @item.listed { item_id, sku: trimmed }
}
