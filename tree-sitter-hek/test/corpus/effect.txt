effect AuditCatalog {
  on @item.listed,
     @item.delisted,
     @item.flagged as e { item_id, seller_id } {

    state listed: Int = fold 0
      on @item.listed(seller_id) => listed + 1

    let response = http.post(
      "https://audit.example/catalog",
      { "seller": seller_id, "listed": listed },
      headers = { "Idempotency-Key": "{audit_id(seller_id, e.position)}" },
    )

    if response.status >= 400 {
      fail("audit rejected")
    }

    log("seller {seller_id} now lists {listed}")

    invoke RecordNotified {
      order_id: e.order_id,
      notification_id: Uuid.derive(e.id, "confirmation"),
    }
  }
}

effect SyncShop {
  on @shop.sync.requested as e { shop_id } {
    state token: String? = fold none
      on @shop.connected(shop_id) { access_token } => access_token

    let secret = reveal(token)
    let at = now()
    if secret.is_none() {
      erase(e.shop_id)
      return
    }
  }
}

effect RecordWarrantySales {
  fn lookup_plan(plans: Map(Uuid, SalePlan), sku: String) -> SalePlan? {
    for plan_id, plan in plans {
      if plan.sku == sku {
        return plan
      }
    }
    return none
  }

  on @shopify.orders_paid.received as e {
    shop_id, order_id, order_name
  } {
    log("order {order_name} for shop {shop_id}")
  }
}

an effect-local fn that declares no result

effect SyncShop {
  fn sync(shop_id: Int, domain: String) {
    let response = http.post("https://{domain}/sync", { "shop": shop_id })
    if response.status >= 400 {
      fail("sync failed")
    }
  }

  on @shop.connected as e { shop_id, shop_domain } {
    sync(shop_id, shop_domain)
  }
}
