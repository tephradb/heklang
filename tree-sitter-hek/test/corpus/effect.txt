effect AuditCatalog {
  on @item.listed,
     @item.delisted,
     @item.flagged as e { item_id, seller_id } {

    state listed: Int = fold 0
      on @item.listed(seller_id) => listed + 1

    let response = http.post(
      "https://audit.example/catalog",
      { "seller": seller_id, "listed": listed },
      headers = { "Idempotency-Key": "{audit_id(seller_id, e.position)}" },
    )

    if response.status >= 400 {
      fail("audit rejected")
    }

    log("seller {seller_id} now lists {listed}")

    invoke RecordNotified {
      order_id: e.order_id,
      notification_id: Uuid.derive(e.id, "confirmation"),
    }
  }
}

effect SyncShop {
  on @shop.sync.requested as e { shop_id } {
    state token: String? = fold none
      on @shop.connected(shop_id) { access_token } => access_token

    let secret = reveal(token)
    let at = now()
    if secret.is_none() {
      erase(e.shop_id)
      return
    }
  }
}
