  expect nothing

test "confirming an order records the notification" {
  given @order.placed { order_id: ADA }

  respond "https://mail.example/confirm" 503
  respond "https://mail.example/confirm" 200 { "ok": true }
  respond "https://mail.example/other" timeout
  erased customer_id "1"

  deliver NotifyCustomer

  expect http.post("https://mail.example/confirm", { "to": "ada@example.com" })
  expect http.get("https://mail.example/confirm")
  expect log("confirmed an order")
  expect fail("confirmation rejected")
  expect erase(customer_id, "1")
  expect invoke RecordNotified { order_id: ADA }
  expect skipped
}
