================================================================================
a named fold declares its accumulator only when an arm reads it
================================================================================

fold OpenProposal(shop_id: Shop, order_id: Int) -> Uuid? = none {
  on @edit.proposed(shop_id, order_id) { proposal_id } => proposal_id
  on @edit.cancelled(shop_id, order_id) => none
}

fold QuoteFailures(order_id: Int) -> failures: Int = 0 {
  on @edit.quote.failed(order_id) => failures + 1
}

fold MerchantEmail(shop_id: Shop) -> Sealed(String, Shop)? = none {
  on @shop.installed(shop_id) { merchant_email } => merchant_email
}

================================================================================
the two `fold` statements are told apart by the token after the name
================================================================================

command ConfirmEdit(shop_id: Shop, order_id: Int) {
  fold open = OpenProposal { shop_id, order_id }
  fold seen: Bool = false
    on @edit.proposed(shop_id, order_id) => true
  emit @edit.confirmed { shop_id, order_id }
}
