================================================================================
a guard declares parameters and folds
================================================================================

guard ShopIsConnected(shop_id: Int) {
  state connected: Bool = fold false
    on @shop.connected(shop_id) => true

  if !connected {
    return reject ShopNotFound
  }
}

================================================================================
raw slices and named guards are told apart by the token after `guard`
================================================================================

command ArchivePlan(plan_id: Uuid, shop_id: Int) {
  guard @plan.archived(plan_id)
  guard ShopIsConnected { shop_id }
  guard PlanExists { plan_id, shop_id: shop_id }

  emit @plan.archived { plan_id, shop_id }
}
