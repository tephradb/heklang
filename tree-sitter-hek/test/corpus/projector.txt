================================================================================
entity with a key, indexes and a default
================================================================================

projector Orders {
  enum Status { @default Placed, Shipped }

  entity Order {
    order_id: Uuid @key,
    customer_id: Int @index,
    email: String @max(200),
    status: Status,
    tracking: String?,
    name: String = "unknown",

    index (customer_id, status)
  }

  on @order.placed as e { order_id, customer_id, email } {
    put Order {
      order_id, customer_id, email,
      status: Placed,
      tracking: none,
      name: e.id,
    }
  }
}

================================================================================
patch, update, delete and a stored read
================================================================================

projector Plans {
  entity Plan { plan_id: Int @key, sold: Int }

  on @plan.sold { plan_id, price } {
    update Plan[plan_id] { sold: .sold + 1 }
    patch Plan[plan_id] { sold: .sold - 1 }
  }

  on @plan.deleted { plan_id } {
    delete Plan[plan_id]
  }
}
